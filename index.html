function doPost(e) { try { // Thay 'chi_tiet_HD' bằng đúng tên trang tính (sheet) của anh nếu khác var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('chi_tiet_HD'); // Lấy dữ liệu từ form web gửi lên var timeString = e.parameter.timeString; var name = e.parameter.name; var phone = e.parameter.phone; var address = e.parameter.address; // <-- LẤY DỮ LIỆU ĐỊA CHỈ NHẬN HÀNG var productCode = e.parameter.productCode; var rentalDate = e.parameter.rentalDate; var returnDate = e.parameter.returnDate; var totalQty = e.parameter.totalQty; var itemsDetail = e.parameter.itemsDetail; var total = e.parameter.total; var deposit = e.parameter.deposit; var extraFee = e.parameter.extraFee; var remaining = e.parameter.remaining; // Ghi dữ liệu vào file Google Sheets (Thứ tự các biến dưới đây chính là thứ tự cột) sheet.appendRow([ timeString, // Cột A: Thời gian name, // Cột B: Họ và tên phone, // Cột C: Số điện thoại address, // Cột D: Địa chỉ nhận hàng (Được chèn ngay sau cột SĐT) productCode, // Cột E: Mã sản phẩm rentalDate, // Cột F: Ngày mượn returnDate, // Cột G: Ngày trả totalQty, // Cột H: Tổng số lượng itemsDetail, // Cột I: Chi tiết đơn total, // Cột J: Tổng tiền deposit, // Cột K: Đặt cọc extraFee, // Cột L: Phụ thu remaining // Cột M: Còn lại ]); return ContentService.createTextOutput(JSON.stringify({"result":"success", "data": "Ghi thành công"})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"result":"error", "error": error.message})) .setMimeType(ContentService.MimeType.JSON); } }
